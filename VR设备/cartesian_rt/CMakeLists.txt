cmake_minimum_required(VERSION 3.16)
project(atom_cartesian_rt LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
find_package(Threads REQUIRED)
set(XCORE_SDK_ROOT "" CACHE PATH "Matched AR SDK root (include/rokae and lib)")
set(XCORE_SDK_LIBRARY "" CACHE FILEPATH "Matching xCoreSDK shared library")
set(XCORE_MODEL_LIBRARY "" CACHE FILEPATH "Optional vendor model archive required by some SDK builds")
set(XCORE_ROBOT_TYPE "ArRobot" CACHE STRING "AR SDK class; older SDK may use xMateErProRobot")
add_executable(cartesian_worker worker.cpp)
target_link_libraries(cartesian_worker PRIVATE Eigen3::Eigen Threads::Threads)
target_compile_options(cartesian_worker PRIVATE -Wall -Wextra -Wpedantic)
if(XCORE_SDK_ROOT)
  if(NOT XCORE_SDK_LIBRARY)
    message(FATAL_ERROR "Set XCORE_SDK_LIBRARY to the matching SDK .so; do not mix SDK releases")
  endif()
  target_include_directories(cartesian_worker PRIVATE "${XCORE_SDK_ROOT}/include")
  target_compile_definitions(cartesian_worker PRIVATE ATOM_WITH_SDK=1 ATOM_ROBOT_TYPE=${XCORE_ROBOT_TYPE})
  target_link_libraries(cartesian_worker PRIVATE "${XCORE_SDK_LIBRARY}")
  get_filename_component(SDK_LIB_DIR "${XCORE_SDK_LIBRARY}" DIRECTORY)
  if(NOT XCORE_MODEL_LIBRARY AND EXISTS "${SDK_LIB_DIR}/libxMateModel.a")
    set(XCORE_MODEL_LIBRARY "${SDK_LIB_DIR}/libxMateModel.a")
  endif()
  if(XCORE_MODEL_LIBRARY)
    target_link_libraries(cartesian_worker PRIVATE "${XCORE_MODEL_LIBRARY}")
  endif()
  set_target_properties(cartesian_worker PROPERTIES BUILD_RPATH "${SDK_LIB_DIR}")
else()
  message(STATUS "Mock-only build: real robot requests will be rejected")
endif()
enable_testing()
add_test(NAME cartesian_self_test COMMAND cartesian_worker --self-test)
